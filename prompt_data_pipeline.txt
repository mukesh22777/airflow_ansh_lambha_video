1. airflow/dags/customer_pipeline.py

from airflow import DAG
from airflow.providers.cncf.kubernetes.operators.pod import KubernetesPodOperator
from airflow.utils.dates import days_ago
from datetime import timedelta

DEFAULT_ARGS = {
    'owner': 'airflow',
    'depends_on_past': False,
    'email_on_failure': False,
    'email_on_retry': False,
    'retries': 2,
    'retry_delay': timedelta(minutes=5)
}

with DAG(
    dag_id='customer_data_pipeline',
    default_args=DEFAULT_ARGS,
    description='Customer ETL Pipeline',
    schedule='30 23 * * 1-5',
    start_date=days_ago(1),
    catchup=False,
    tags=['customer', 'spark', 'kubernetes']
) as dag:

    extract_task = KubernetesPodOperator(
        namespace='default',
        image='customer-pipeline:latest',
        cmds=['python'],
        arguments=['/app/spark_jobs/extract.py'],
        labels={'app': 'customer-pipeline'},
        name='extract-task',
        task_id='extract_task',
        get_logs=True,
        is_delete_operator_pod=True
    )

    bronze_task = KubernetesPodOperator(
        namespace='default',
        image='customer-pipeline:latest',
        cmds=['python'],
        arguments=['/app/spark_jobs/bronze_load.py'],
        labels={'app': 'customer-pipeline'},
        name='bronze-task',
        task_id='bronze_task',
        get_logs=True,
        is_delete_operator_pod=True
    )

    silver_task = KubernetesPodOperator(
        namespace='default',
        image='customer-pipeline:latest',
        cmds=['python'],
        arguments=['/app/spark_jobs/silver_transform.py'],
        labels={'app': 'customer-pipeline'},
        name='silver-task',
        task_id='silver_task',
        get_logs=True,
        is_delete_operator_pod=True
    )

    gold_task = KubernetesPodOperator(
        namespace='default',

        cmds=['python'],
        arguments=['/app/spark_jobs/gold_load.py'],
        labels={'app': 'customer-pipeline'},
        name='gold-task',

    )

    extract_task >> bronze_task >> silver_task >> gold_task

2. spark_jobs/extract.py

from pyspark.sql import SparkSession
import json

spark = SparkSession.builder \
    .appName('CustomerExtract') \
    .enableHiveSupport() \
    .getOrCreate()

with open('/app/configs/app_config.json') as config_file:
    config = json.load(config_file)

jdbc_url = config['sqlserver']['jdbc_url']
user = config['sqlserver']['user']
password = config['sqlserver']['password']
source_table = config['sqlserver']['source_table']

query = f"(SELECT * FROM {source_table}) as customer_data"

customer_df = spark.read \
    .format('jdbc') \
    .option('url', jdbc_url) \
    .option('dbtable', query) \
    .option('user', user) \
    .option('password', password) \
    .option('driver', 'com.microsoft.sqlserver.jdbc.SQLServerDriver') \
    .load()

customer_df.write.mode('overwrite').parquet('/data/raw/customer/')

print('Data extraction completed successfully')

3. spark_jobs/bronze_load.py

from pyspark.sql import SparkSession
from pyspark.sql.functions import current_timestamp

spark = SparkSession.builder \
    .appName('BronzeLoad') \
    .enableHiveSupport() \
    .getOrCreate()

raw_df = spark.read.parquet('/data/raw/customer/')

bronze_df = raw_df.withColumn('ingestion_timestamp', current_timestamp())

bronze_df.write \
    .mode('overwrite') \
    .partitionBy('GeographyKey') \
    .parquet('/data/bronze/customer/')

print('Bronze layer load completed')

4. spark_jobs/silver_transform.py

from pyspark.sql import SparkSession
from pyspark.sql.functions import col, upper

spark = SparkSession.builder \
    .appName('SilverTransform') \
    .enableHiveSupport() \
    .getOrCreate()

bronze_df = spark.read.parquet('/data/bronze/customer/')

silver_df = bronze_df.dropDuplicates(['CustomerKey']) \
    .filter(col('EmailAddress').isNotNull()) \
    .withColumn('FirstName', upper(col('FirstName'))) \
    .withColumn('LastName', upper(col('LastName')))

silver_df.write \
    .mode('overwrite') \
    .partitionBy('GeographyKey') \
    .parquet('/data/silver/customer/')

print('Silver transformation completed')

5. spark_jobs/gold_load.py

from pyspark.sql import SparkSession
from pyspark.sql.functions import col

spark = SparkSession.builder \
    .appName('GoldLoad') \
    .enableHiveSupport() \
    .getOrCreate()

silver_df = spark.read.parquet('/data/silver/customer/')

gold_df = silver_df.filter(col('YearlyIncome') > 50000)

# Load into Hive

gold_df.write \
    .mode('overwrite') \
    .partitionBy('GeographyKey') \
    .saveAsTable('customer_gold')

# Load into S3

gold_df.write \
    .mode('overwrite') \
    .partitionBy('GeographyKey') \
    .parquet('s3a://customer-data/gold/customer/')

print('Gold layer load completed')

    CustomerKey INT,
    GeographyKey INT,
    CustomerAlternateKey STRING,
    FirstName STRING,
    LastName STRING,
    EmailAddress STRING,
    YearlyIncome DOUBLE
)
PARTITIONED BY (GeographyKey INT)
STORED AS PARQUET
LOCATION '/warehouse/customer_gold';
